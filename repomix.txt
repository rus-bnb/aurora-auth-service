const requestToPromise = (req) =>
    new Promise((resolve, reject) => {
        req.onsuccess = () => resolve(req.result);
        req.onerror = () => reject(req.error);
    });

export class Database {
    #name;
    #version;
    #schema;
    #db = null;
    #openPromise = null;

    constructor({ name, schema }) {
        if (!name) throw new Error('Database name is required');
        if (!schema?.stores) throw new Error('Schema with stores is required');
        this.#name = name;
        this.#schema = schema;
        this.#version = schema.version ?? 1;
    }

    get isOpen() {
        return this.#db !== null;
    }

    async open() {
        if (this.#db) return this.#db;
        if (this.#openPromise) return this.#openPromise;

        this.#openPromise = new Promise((resolve, reject) => {
            const request = indexedDB.open(this.#name, this.#version);

            request.onupgradeneeded = (event) => {
                this.#applySchema(request.result, request.transaction, event.oldVersion);
            };

            request.onsuccess = () => {
                this.#db = request.result;
                this.#db.onversionchange = () => {
                    this.#db?.close();
                    this.#db = null;
                };
                resolve(this.#db);
            };

            request.onerror = () => reject(request.error);
            request.onblocked = () =>
                reject(new Error('Database upgrade blocked by another connection'));
        }).finally(() => {
            this.#openPromise = null;
        });

        return this.#openPromise;
    }

    close() {
        this.#db?.close();
        this.#db = null;
    }

    async transaction(storeNames, mode, executor) {
        const db = await this.open();
        const names = Array.isArray(storeNames) ? storeNames : [storeNames];

        return new Promise((resolve, reject) => {
            const tx = db.transaction(names, mode);
            let execResult;
            let execError;

            tx.oncomplete = () => (execError ? reject(execError) : resolve(execResult));
            tx.onerror = () => reject(tx.error);
            tx.onabort = () =>
                reject(execError ?? tx.error ?? new Error('Transaction aborted'));

            try {
                execResult = executor(tx);
                if (execResult && typeof execResult.then === 'function') {
                    execResult.catch((e) => {
                        execError = e;
                        try { tx.abort(); } catch { /* already aborted */ }
                    });
                }
            } catch (e) {
                execError = e;
                try { tx.abort(); } catch { /* already aborted */ }
            }
        });
    }

    #run(storeName, mode, fn) {
        return this.transaction(storeName, mode, (tx) =>
            fn(tx.objectStore(storeName)),
        );
    }

    get(storeName, key) { return this.#run(storeName, 'readonly', (s) => requestToPromise(s.get(key))); }
    getAll(storeName, q, n) { return this.#run(storeName, 'readonly', (s) => requestToPromise(s.getAll(q, n))); }
    getByIndex(storeName, i, v) { return this.#run(storeName, 'readonly', (s) => requestToPromise(s.index(i).get(v))); }
    getAllByIndex(storeName, i, v) { return this.#run(storeName, 'readonly', (s) => requestToPromise(s.index(i).getAll(v))); }
    add(storeName, v) { return this.#run(storeName, 'readwrite', (s) => requestToPromise(s.add(v))); }
    put(storeName, v) { return this.#run(storeName, 'readwrite', (s) => requestToPromise(s.put(v))); }
    delete(storeName, key) { return this.#run(storeName, 'readwrite', (s) => requestToPromise(s.delete(key))); }
    count(storeName, q) { return this.#run(storeName, 'readonly', (s) => requestToPromise(s.count(q))); }
    clear(storeName) { return this.#run(storeName, 'readwrite', (s) => requestToPromise(s.clear())); }

    #applySchema(db, tx, _oldVersion) {
        for (const [storeName, def] of Object.entries(this.#schema.stores)) {
            let store;
            if (!db.objectStoreNames.contains(storeName)) {
                store = db.createObjectStore(storeName, {
                    keyPath: def.keyPath,
                    autoIncrement: def.autoIncrement ?? false,
                });
            } else {
                store = tx.objectStore(storeName);
            }
            for (const [indexName, idxDef] of Object.entries(def.indexes ?? {})) {
                if (!store.indexNames.contains(indexName)) {
                    store.createIndex(indexName, idxDef.keyPath, {
                        unique: idxDef.unique ?? false,
                        multiEntry: idxDef.multiEntry ?? false,
                    });
                }
            }
        }
    }
}

import { hashPassword, verifyPassword } from '../utils/password.js';
import { uuid } from '../utils/id.js';
import { AuthError } from './errors.js';
import { EmailService } from './EmailService.js';

const THIRTY_DAYS = 30 * 24 * 60 * 60 * 1000;
import { EmailService } from './EmailService.js';

const TOKEN_TTL = 24 * 60 * 60 * 1000; // 24 часа

export class AuthService {
    #userRepo;
    #sessionRepo;
    #verificationRepo;
    #passwordResetRepo;
    #emailService;
    #currentUser = null;

    constructor({ userRepo, sessionRepo, verificationRepo, passwordResetRepo, emailService }) {
        this.#userRepo = userRepo;
        this.#sessionRepo = sessionRepo;
        this.#verificationRepo = verificationRepo;
        this.#passwordResetRepo = passwordResetRepo;
        this.#emailService = emailService;
    }

    async register({ name, email, password }) {
        if (!email || !password) throw new AuthError('Заполните все поля');

        const exists = await this.#userRepo.findByEmail(email);
        if (exists) throw new AuthError('Аккаунт с таким email уже существует');

        const user = {
            id: uuid(),
            name: name?.trim() || email,
            email: email.toLowerCase(),
            passwordHash: await hashPassword(password),
            emailVerified: false,
            createdAt: Date.now(),
        };

        await this.#userRepo.add(user);
        await this.#sendVerification(user);
        await this.#sessionRepo.save({ userId: user.id, createdAt: Date.now(), expiresAt: Date.now() + THIRTY_DAYS });
        this.#currentUser = this.#publicView(user);
        return this.#currentUser;
    }

    async #sendVerification(user) {
        await this.#verificationRepo.deleteByUser?.(user.id); // если добавим метод
        const token = uuid();
        await this.#verificationRepo.add({
            token,
            userId: user.id,
            expiresAt: Date.now() + TOKEN_TTL,
        });
        const link = `${location.origin}${location.pathname}#/verify?token=${token}`;
        await this.#emailService.sendVerification(user.email, link);
    }

    async verifyEmail(token) {
        const record = await this.#verificationRepo.findById(token);
        if (!record) throw new AuthError('Ссылка недействительна или истекла');
        if (record.expiresAt < Date.now()) {
            await this.#verificationRepo.remove(token);
            throw new AuthError('Ссылка истекла');
        }

        const user = await this.#userRepo.findById(record.userId);
        if (!user) throw new AuthError('Пользователь не найден');

        user.emailVerified = true;
        await this.#userRepo.save(user);
        await this.#verificationRepo.remove(token);

        if (this.#currentUser?.id === user.id) {
            this.#currentUser = this.#publicView(user);
        }
        return this.#currentUser;
    }

    async resendVerification() {
        if (!this.#currentUser) throw new AuthError('Требуется вход');
        const user = await this.#userRepo.findById(this.#currentUser.id);
        if (!user) throw new AuthError('Пользователь не найден');
        if (user.emailVerified) throw new AuthError('Email уже подтверждён');
        await this.#sendVerification(user);
    }

    async requestPasswordReset(email) {
        const user = await this.#userRepo.findByEmail(email);
        if (!user) throw new AuthError('Если аккаунт существует, письмо отправлено');
        if (!user.emailVerified) {
            throw new AuthError('Сначала подтвердите email — восстановление пароля недоступно');
        }

        const token = uuid();
        await this.#passwordResetRepo.add({
            token,
            userId: user.id,
            expiresAt: Date.now() + TOKEN_TTL,
        });
        const link = `${location.origin}${location.pathname}#/reset-password?token=${token}`;
        await this.#emailService.sendPasswordReset(user.email, link);
    }

    async resetPassword(token, newPassword) {
        const record = await this.#passwordResetRepo.findById(token);
        if (!record) throw new AuthError('Ссылка недействительна или истекла');
        if (record.expiresAt < Date.now()) {
            await this.#passwordResetRepo.remove(token);
            throw new AuthError('Ссылка истекла');
        }

        const user = await this.#userRepo.findById(record.userId);
        if (!user) throw new AuthError('Пользователь не найден');

        user.passwordHash = await hashPassword(newPassword);
        await this.#userRepo.save(user);
        await this.#passwordResetRepo.remove(token);
        await this.#sessionRepo.remove(user.id); // разлогинить везде
    }

    async updateProfile(userId, { name, email }) {
        const user = await this.#userRepo.findById(userId);
        if (!user) throw new AuthError('Пользователь не найден');

        const nextEmail = email ? email.toLowerCase() : user.email;
        if (nextEmail !== user.email) {
            const exists = await this.#userRepo.findByEmail(nextEmail);
            if (exists) throw new AuthError('Этот email уже занят');
            user.email = nextEmail;
            user.emailVerified = false;
        }
        if (name != null) user.name = name.trim() || user.name;

        await this.#userRepo.save(user);

        if (!user.emailVerified) {
            await this.#sendVerification(user);
        }

        if (this.#currentUser?.id === user.id) {
            this.#currentUser = this.#publicView(user);
        }
        return this.#currentUser;
    }

    #publicView({ passwordHash, ...safe }) { return safe; }
}

const EMAILJS_SERVICE_ID  = 'YOUR_SERVICE_ID';
const EMAILJS_PUBLIC_KEY  = 'YOUR_PUBLIC_KEY';
const TEMPLATE_VERIFY     = 'template_verify';
const TEMPLATE_RESET      = 'template_reset';

export class EmailService {
    #ready = false;

    #ensureReady() {
        if (this.#ready) return;
        if (!window.emailjs) throw new Error('EmailJS SDK не загружен');
        emailjs.init(EMAILJS_PUBLIC_KEY);
        this.#ready = true;
    }

    async sendVerification(email, link) {
        this.#ensureReady();
        return emailjs.send(EMAILJS_SERVICE_ID, TEMPLATE_VERIFY, {
            to_email: email,
            verification_link: link,
        });
    }

    async sendPasswordReset(email, link) {
        this.#ensureReady();
        return emailjs.send(EMAILJS_SERVICE_ID, TEMPLATE_RESET, {
            to_email: email,
            reset_link: link,
        });
    }
}

export class AuthForm extends HTMLElement {

        this.#loading = on;

        if (!btn) return;
        btn.disabled = on;
        btn.classList.toggle('is-loading', on);
    }

    #togglePassword(btn) {
        const input = btn.parentElement.querySelector('input');
        const show = input.type === 'password';
        input.type = show ? 'text' : 'password';
        btn.innerHTML = show ? ICONS.eyeOff : ICONS.eye;
        btn.setAttribute('aria-pressed', String(show));
        btn.setAttribute('aria-label', show ? 'Скрыть пароль' : 'Показать пароль');
        input.focus({ preventScroll: true });
    }

    #onInput(e) {
        const input = e.target;
        if (!(input instanceof HTMLInputElement)) return;
        const wrap = this.#fieldOf(input);
        const field = this.#fieldSchema(input.name);
        if (!wrap || !field) return;

        if (wrap.classList.contains('is-invalid')) {
            const value = input.type === 'checkbox' ? input.checked : input.value;
            this.#setError(wrap, validateField(field, value, this.#values()));

        const value = input.type === 'checkbox' ? input.checked : input.value;

        if (!meter) return;

        let score = 0;
        if (value.length >= 8) score++;
        if (value.length >= 12) score++;
        if (/[A-ZА-ЯЁ]/.test(value) && /[a-zа-яё]/.test(value)) score++;
        if (/\d/.test(value)) score++;
        if (/[^\p{L}\p{N}\s]/u.test(value)) score++;
        score = Math.min(score, 4);
        if (!value) score = 0;

        meter.dataset.level = String(score);
        if (label) label.textContent = value ? STRENGTH_LABELS[score] : '';
    }

    #runAction(action) {
        if (action === 'forgot') {
            const emailInput = this.querySelector('input[name="email"]');
            const email = emailInput?.value.trim() ?? '';

            if (!EMAIL_RE.test(email)) {
                this.#formError('Укажите email');
                emailInput?.focus();
                return;
            }

            this.#formError('');
            this.#setNote('Локальная демо-версия не поддерживает восстановление пароля.');
        }
    }

    async #onSubmit(e) {
        e.preventDefault();
        if (this.#loading) return;

        this.#formError('');
        this.#setNote('');

        const values = this.#values();
        let firstInvalid = null;

        for (const field of this.#schema.fields) {
            const wrap = this.querySelector(`[data-field="${field.name}"]`);
            if (!wrap) continue;
            const error = validateField(field, values[field.name], values);
            this.#setError(wrap, error);
            if (error && !firstInvalid) firstInvalid = wrap.querySelector('input');
        }

        if (firstInvalid) {
            firstInvalid.focus();
            this.#formError('Проверьте выделенные поля');
            return;
        }

        this.#setLoading(true);
        try {
            const user = await this.#schema.onSubmit(values);
            this.#setNote(this.#schema.successText || 'Готово!');
            this.dispatchEvent(new CustomEvent('auth:success', {
                detail: { user },
                bubbles: true,
                composed: true,
            }));
        } catch (err) {
            this.#formError(err?.message || 'Не удалось выполнить запрос');
        } finally {
            this.#setLoading(false);
        }
    }
}

export function uuid() {
    if (typeof crypto !== 'undefined' && crypto.randomUUID) return crypto.randomUUID();
    const b = crypto.getRandomValues(new Uint8Array(16));
    b[6] = (b[6] & 0x0f) | 0x40;
    b[8] = (b[8] & 0x3f) | 0x80;
    const h = [...b].map((x) => x.toString(16).padStart(2, '0')).join('');
    return `${h.slice(0, 8)}-${h.slice(8, 12)}-${h.slice(12, 16)}-${h.slice(16, 20)}-${h.slice(20)}`;
}

const enc = new TextEncoder();
const toHex = (buf) =>
    [...new Uint8Array(buf)].map((b) => b.toString(16).padStart(2, '0')).join('');
const fromHex = (hex) =>
    new Uint8Array(hex.match(/.{2}/g).map((b) => parseInt(b, 16)));

const derive = async (password, salt) => {
    const key = await crypto.subtle.importKey(
        'raw', enc.encode(password), 'PBKDF2', false, ['deriveBits'],
    );
    return crypto.subtle.deriveBits(
        { name: 'PBKDF2', salt, iterations: 100_000, hash: 'SHA-256' },
        key, 256,
    );
};

export async function hashPassword(password) {
    const salt = crypto.getRandomValues(new Uint8Array(16));
    const bits = await derive(password, salt);
    return `${toHex(salt)}:${toHex(bits)}`;
}

export async function verifyPassword(password, stored) {
    const [saltHex, hashHex] = stored.split(':');
    const bits = await derive(password, fromHex(saltHex));
    const computed = toHex(bits);
    if (computed.length !== hashHex.length) return false;
    let diff = 0;
    for (let i = 0; i < computed.length; i++) {
        diff |= computed.charCodeAt(i) ^ hashHex.charCodeAt(i);
    }
    return diff === 0;
}

export const EMAIL_RE = /^[^\s@]+@[^\s@]+\.[^\s@]{2,}$/;

export function validateField(field, value, allValues) {
    const rules = field.rules || {};

    if (field.type === 'checkbox') {
        return rules.required && !value ? rules.required : '';
    }

    const v = String(value ?? '').trim();

    if (!v) return rules.required || '';
    if (rules.min && v.length < rules.min) return rules.minMsg || `Минимум ${rules.min} символов`;
    if (rules.email && !EMAIL_RE.test(v)) return 'Некорректный email';

    if (rules.match) {
        const other = String(allValues[rules.match.field] ?? '');
        if (v !== other) return rules.match.msg || 'Значения не совпадают';
    }

    return '';
}

function parseRoute() {
    const raw = location.hash.replace(/^#\/?/, '');
    const [path, query = ''] = raw.split('?');
    return { path: path || 'login', params: new URLSearchParams(query) };
}

function renderProfile() {
    const u = auth.currentUser;
    if (!u) return mountLogin('login');
    view.innerHTML = '';
    const el = document.createElement('profile-view');
    el.data = { user: u, auth };
    view.appendChild(el);
    restartAnim();
}

function renderSimple(tag, props) {
    view.innerHTML = '';
    const el = document.createElement(tag);
    if (props) Object.assign(el, props);
    view.appendChild(el);
    restartAnim();
}

function route() {
    const { path, params } = parseRoute();

    switch (path) {
        case 'profile':
            renderProfile();
            break;
        case 'forgot':
            renderSimple('forgot-password-view', { auth });
            break;
        case 'reset-password':

            break;
        case 'verify':

            break;
        case 'register':
        case 'login':
